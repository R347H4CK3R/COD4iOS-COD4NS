cmake_minimum_required(VERSION 3.20)
project(IW5SnapshotCore VERSION 0.1.0 LANGUAGES C)
add_library(iw5geo STATIC src/iw5geo.c)
target_include_directories(iw5geo PUBLIC include)
target_compile_features(iw5geo PUBLIC c_std_11)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(iw5geo PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    include(CTest)
    add_executable(iw5geo-inspect tools/iw5geo_inspect.c)
    target_link_libraries(iw5geo-inspect PRIVATE iw5geo)
    if(BUILD_TESTING)
        add_executable(iw5geo-tests tests/iw5geo_tests.c)
        target_link_libraries(iw5geo-tests PRIVATE iw5geo)
        add_test(NAME iw5geo-boundaries COMMAND iw5geo-tests)
    endif()
endif()
