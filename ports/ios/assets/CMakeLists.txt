cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(KisakCODAssets LANGUAGES CXX)
    include(CTest)
endif()

enable_language(C)
if(NOT TARGET iw5geo)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../iw5-native-core/native" "${CMAKE_CURRENT_BINARY_DIR}/iw5-native-core")
endif()
find_package(ZLIB REQUIRED)
add_library(kisakcod_assets STATIC BspLoader.cpp FastfileWorld.cpp IW5Snapshot.cpp)
target_link_libraries(kisakcod_assets PUBLIC ZLIB::ZLIB iw5geo)
target_compile_features(kisakcod_assets PUBLIC cxx_std_17)
target_include_directories(kisakcod_assets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(kisakcod_assets PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
    add_executable(iw5_snapshot_tests tests/IW5SnapshotTests.cpp)
    target_link_libraries(iw5_snapshot_tests PRIVATE kisakcod_assets)
    add_test(NAME iw5_snapshot_adapter COMMAND iw5_snapshot_tests)
    add_executable(kisakcod_assets_tests tests/BspLoaderTests.cpp)
    target_link_libraries(kisakcod_assets_tests PRIVATE kisakcod_assets)
    add_test(NAME kisakcod_assets_tests COMMAND kisakcod_assets_tests)
    add_executable(kisakcod_map_inspect MapInspect.cpp)
    target_link_libraries(kisakcod_map_inspect PRIVATE kisakcod_assets)
    add_executable(kisakcod_fastfile_tests tests/FastfileWorldTests.cpp)
    target_link_libraries(kisakcod_fastfile_tests PRIVATE kisakcod_assets)
    add_test(NAME kisakcod_fastfile_tests COMMAND kisakcod_fastfile_tests)
endif()
